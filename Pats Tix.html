import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# LOAD & PREP DATA
df = pd.read_excel(r"C:\Users\student\OneDrive - Bryant University\Desktop\Patriots Ticket Sales.xlsx")

df.columns = ["Year", "Avg Resale Ticket Price", "Avg Attendance %", "Team Payroll", "Win %"]

# Exclude "Year"
df_analysis = df[["Avg Resale Ticket Price", "Avg Attendance %", "Team Payroll", "Win %"]]

# CORRELATION MATRIX
corr = df_analysis.corr(numeric_only=True)

print("=== CORRELATION MATRIX ===")
print(corr, "\n")

# CORRELATION HEATMAP
plt.figure(figsize=(8,6))
sns.heatmap(corr, annot=True, cmap="coolwarm", fmt=".2f", linewidths=0.5)
plt.title("Correlation Heatmap Between Team Metrics", fontsize=14, pad=10)
plt.tight_layout()
plt.show()

# SCATTERPLOTS 
sns.set(style="whitegrid")
vars_to_compare = list(df_analysis.columns)

for i, x_var in enumerate(vars_to_compare):
    for j, y_var in enumerate(vars_to_compare):
        if i < j:  # Avoid duplicates and self-comparisons
            plt.figure(figsize=(6,4))
            sns.regplot(
                data=df_analysis, x=x_var, y=y_var,
                scatter_kws={'alpha':0.7, 's':50},
                line_kws={'color':'red'}
            )
            r = corr.loc[x_var, y_var]
            plt.title(f"{y_var} vs {x_var}\nCorrelation: {r:.2f}", fontsize=11)
            plt.tight_layout()
            plt.show()

# CORRELATIONS SUMMARY
corr_pairs = corr.unstack().reset_index()
corr_pairs.columns = ["Variable 1", "Variable 2", "Correlation"]

# Remove duplicates
corr_pairs = corr_pairs[corr_pairs["Variable 1"] != corr_pairs["Variable 2"]]
corr_pairs = corr_pairs.drop_duplicates(subset=["Correlation"], keep="first")

# Sort
corr_pairs["Abs Corr"] = corr_pairs["Correlation"].abs()
corr_pairs = corr_pairs.sort_values(by="Abs Corr", ascending=False)

print("=== STRONGEST CORRELATIONS ===")
print(corr_pairs.head(10), "\n")

top_pair = corr_pairs.iloc[0]
v1, v2, r_val = top_pair["Variable 1"], top_pair["Variable 2"], top_pair["Correlation"]

plt.figure(figsize=(8,6))
sns.heatmap(corr, annot=True, cmap="coolwarm", fmt=".2f", linewidths=0.5)
plt.title(f"Strongest Correlation: {v1} ↔ {v2} (r = {r_val:.2f})", fontsize=13, pad=10)
plt.tight_layout()
plt.show()
